### Файл app/Http/Controllers/HomeController.php

```php
<?php

namespace App\Http\Controllers;

use App\Models\Part;
use App\Models\Category;
use App\Services\PartCacheService;

class HomeController extends Controller
{
    protected PartCacheService $cacheService;

    public function __construct(PartCacheService $cacheService)
    {
        $this->cacheService = $cacheService;
    }

    public function index()
    {
        // Популярные товары
        $popularParts = Part::where('is_active', true)
            ->orderBy('views', 'desc')
            ->limit(12)
            ->get();

        // Новые товары
        $newParts = Part::where('is_active', true)
            ->orderBy('created_at', 'desc')
            ->limit(12)
            ->get();

        // Категории
        $categories = Category::all();

        // Акции
        $saleParts = Part::where('is_active', true)
            ->where('discount', '>', 0)
            ->orderBy('discount', 'desc')
            ->limit(8)
            ->get();

        return view('home', compact('popularParts', 'newParts', 'categories', 'saleParts'));
    }
}
```
